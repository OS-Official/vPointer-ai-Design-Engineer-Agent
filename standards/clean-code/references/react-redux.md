## 1. Derive, don't store

```jsx
// ✗ Two sources of truth that can drift apart
const [models, setModels] = useState(all);
const [visible, setVisible] = useState([]);
useEffect(() => setVisible(models.filter((m) => m.capacity >= min)), [models, min]);

// ✓ One source, derived during render
const visible = models.filter((m) => m.capacity >= min);
// Only if the list is large and re-renders are measured to be slow:
const visible = useMemo(() => models.filter((m) => m.capacity >= min), [models, min]);
```

## 2. Effect with cleanup

```jsx
useEffect(() => {
  const media = window.matchMedia('(min-width: 64rem)');
  const update = () => setIsDesktop(media.matches);
  update();
  media.addEventListener('change', update);
  return () => media.removeEventListener('change', update); // cleanup: no leaks, no double listeners in Strict Mode
}, []);
```
Layout belongs in CSS. Use a JS media query only when *behaviour* changes, such as mounting a different component. Never use one to change styles.

## 3. Four states of a data view

```jsx
function ModelList() {
  const { data: models, isLoading, isError, refetch } = useGetModelsQuery();
  if (isLoading) return <p role="status">Loading models…</p>;
  if (isError) return (
    <div role="alert">
      <p>We couldn't load the models.</p>
      <button type="button" onClick={refetch}>Try again</button>
    </div>
  );
  if (!models?.length) return <p>No models match these filters.</p>;
  return <ul role="list">{models.map((m) => <ModelCard key={m.id} model={m} />)}</ul>;
}
```

## 4. Redux Toolkit slice + memoised selector

```js
// features/models/modelsSlice.js
import { createSlice, createSelector } from '@reduxjs/toolkit';

const modelsSlice = createSlice({
  name: 'models',
  initialState: { selectedId: 'trail-pack-30', minCapacity: 0 },
  reducers: {
    modelSelected(state, action) { state.selectedId = action.payload; },   // Immer makes this safe
    minCapacityChanged(state, action) { state.minCapacity = action.payload; },
  },
});

export const { modelSelected, minCapacityChanged } = modelsSlice.actions;
export default modelsSlice.reducer;

export const selectSelectedId = (state) => state.models.selectedId;
export const selectMinCapacity = (state) => state.models.minCapacity;
// Memoised: recomputes only when its inputs change, so components don't re-render for nothing.
export const makeSelectVisible = (allModels) =>
  createSelector([selectMinCapacity], (min) => allModels.filter((m) => m.capacity >= min));
```

```js
// app/store.js
import { configureStore } from '@reduxjs/toolkit';
import models from '../features/models/modelsSlice';
import { api } from '../services/api';

export const store = configureStore({
  reducer: { models, [api.reducerPath]: api.reducer },
  middleware: (getDefault) => getDefault().concat(api.middleware),
});
```

## 5. RTK Query for server data

```js
// services/api.js
import { createApi, fetchBaseQuery } from '@reduxjs/toolkit/query/react';
export const api = createApi({
  baseQuery: fetchBaseQuery({ baseUrl: '/api' }),
  endpoints: (b) => ({ getModels: b.query({ query: () => 'models' }) }),
});
export const { useGetModelsQuery } = api;
```
Why: caching, loading and error flags, and de-duplication of requests come built in, so there's no hand-written `isLoading` in a slice.
